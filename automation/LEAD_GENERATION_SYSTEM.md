### **Lead Capture Automation**
```javascript
// Enhanced website form with lead scoring
function captureAndScoreLead(formData) {
  let leadScore = 0;
  
  // Project type scoring
  if (formData.projectType === 'commercial') leadScore += 30;
  if (formData.projectType === 'basement') leadScore += 20;
  if (formData.projectType === 'new construction') leadScore += 25;
  
  // Timeline scoring  
  if (formData.timeline === 'this month') leadScore += 25;
  if (formData.timeline === 'next 3 months') leadScore += 15;
  
  // Budget indicators
  if (formData.sqft > 1000) leadScore += 20;
  
  return {
    ...formData,
    leadScore,
    priority: leadScore >= 40 ? 'high' : leadScore >= 20 ? 'medium' : 'low'
  };
}
```
